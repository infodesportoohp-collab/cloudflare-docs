// routes/login.tsx
import { Form, redirect } from "@remix-run/react";
import { validarCodigo } from "~/lib/auth.server";

export async function action({ request }) {
  const form = await request.formData();
  const codigo = form.get("codigo");

  const valido = await validarCodigo(codigo);
  if (!valido) {
    return { error: "Código inválido ou já usado" };
  }

  return redirect("/votacao");
}

export default function Login() {
  return (
    <Form method="post">
      <input name="codigo" placeholder="Código de acesso" />
      <button type="submit">Entrar</button>
    </Form>
  );
}
